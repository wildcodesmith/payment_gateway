  <script>
    // ========== INTERACTIVITY ENGINE ==========
    const form = document.getElementById('form');
    const fields = {
      name: document.getElementById('userName'),
      roll: document.getElementById('userRollNumber'),
      event: document.getElementById('userEvent'),
      mail: document.getElementById('userMail')
    };
    const progressFill = document.getElementById('progressFill');
    const progressLabel = document.getElementById('progressLabel');
    const nameCount = document.getElementById('nameCount');
    const payBtn = document.getElementById('payBtn');
    const btnText = document.getElementById('btnText');
    const chargeBar = document.getElementById('chargeBar');
    const statusText = document.getElementById('statusText');
    const cursorGlow = document.getElementById('cursorGlow');
    const sparkGroup = document.getElementById('sparkGroup');

    // ---- Cursor glow ----
    let mouseX = 0, mouseY = 0;
    document.addEventListener('mousemove', (e) => {
      mouseX = e.clientX;
      mouseY = e.clientY;
      cursorGlow.style.left = mouseX + 'px';
      cursorGlow.style.top = mouseY + 'px';
      cursorGlow.style.opacity = '1';
    });
    document.addEventListener('mouseleave', () => {
      cursorGlow.style.opacity = '0';
    });

    // ---- Field validation & progress ----
    function isValid(field) {
      if (field === fields.name) return field.value.trim().length >= 2;
      if (field === fields.roll) return field.value.trim().length >= 3;
      if (field === fields.event) return field.value !== '';
      if (field === fields.mail) return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(field.value);
      return false;
    }

    function updateProgress() {
      let valid = 0;
      Object.values(fields).forEach(f => {
        if (isValid(f)) {
          valid++;
          f.classList.add('field-valid');
          f.classList.remove('field-invalid');
        } else if (f.value.length > 0) {
          f.classList.add('field-invalid');
          f.classList.remove('field-valid');
        } else {
          f.classList.remove('field-valid', 'field-invalid');
        }
      });
      const pct = (valid / 4) * 100;
      progressFill.style.width = pct + '%';
      progressLabel.textContent = `${valid} / 4`;
      if (valid === 4) {
        statusText.textContent = 'ALL NODES LOCKED • READY';
        statusText.classList.add('text-cyan-300');
      } else {
        statusText.textContent = 'BUS_CLK: 4.8GHZ';
        statusText.classList.remove('text-cyan-300');
      }
    }

    fields.name.addEventListener('input', () => {
      nameCount.textContent = fields.name.value.length;
      updateProgress();
    });
    Object.values(fields).forEach(f => {
      f.addEventListener('input', updateProgress);
      f.addEventListener('change', updateProgress);
      f.addEventListener('blur', updateProgress);
    });

    // ---- Pin flicker on focus ----
    document.querySelectorAll('input, select').forEach(el => {
      el.addEventListener('focus', () => {
        document.querySelectorAll('.pin').forEach((p, i) => {
          setTimeout(() => {
            p.style.boxShadow = '0 0 12px #22d3ee';
            p.style.background = '#22d3ee';
            setTimeout(() => {
              p.style.boxShadow = '';
              p.style.background = '';
            }, 300);
          }, i * 40);
        });
      });
    });

    // ---- Charge-to-pay button ----
    let charge = 0;
    let charging = false;
    let chargeInterval = null;

    function startCharge() {
      if (Object.values(fields).every(isValid) === false) {
        // shake invalid fields
        Object.values(fields).forEach(f => {
          if (!isValid(f)) {
            f.classList.add('field-invalid');
            f.animate([
              { transform: 'translateX(0)' },
              { transform: 'translateX(-6px)' },
              { transform: 'translateX(6px)' },
              { transform: 'translateX(0)' }
            ], { duration: 300, iterations: 2 });
          }
        });
        statusText.textContent = 'INCOMPLETE PARAMETERS';
        return;
      }
      charging = true;
      btnText.textContent = 'CHARGING…';
      chargeInterval = setInterval(() => {
        charge = Math.min(100, charge + 2.2);
        chargeBar.style.width = charge + '%';
        if (charge >= 100) {
          clearInterval(chargeInterval);
          btnText.textContent = 'READY — RELEASE TO PAY';
          payBtn.style.borderColor = '#67e8f9';
          payBtn.style.boxShadow = '0 0 40px rgba(34,211,238,1)';
        }
      }, 30);
    }

    function endCharge() {
      if (!charging) return;
      clearInterval(chargeInterval);
      if (charge >= 100) {
        // success burst
        createBurst(payBtn);
        btnText.textContent = 'INITIATING PAYMENT…';
        statusText.textContent = 'TRANSACTION NODE OPEN';
        // Call your existing Razorpay / formScript logic here

    // ---- Particle burst ----
    function createBurst(el) {
      const rect = el.getBoundingClientRect();
      const cx = rect.left + rect.width / 2;
      const cy = rect.top + rect.height / 2;
      for (let i = 0; i < 28; i++) {
        const p = document.createElement('div');
        p.className = 'particle';
        document.body.appendChild(p);
        const angle = (Math.PI * 2 * i) / 28;
        const dist = 40 + Math.random() * 80;
        p.style.left = cx + 'px';
        p.style.top = cy + 'px';
        p.animate([
          { transform: 'translate(0,0) scale(1)', opacity: 1 },
          { transform: `translate(${Math.cos(angle)*dist}px, ${Math.sin(angle)*dist}px) scale(0)`, opacity: 0 }
        ], { duration: 700 + Math.random()*300, easing: 'cubic-bezier(0.1,0.8,0.2,1)' });
        setTimeout(() => p.remove(), 1000);
      }
    }
